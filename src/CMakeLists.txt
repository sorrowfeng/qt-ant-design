# Visual Studio injects /RTC1 into Debug, while MSVC AddressSanitizer is not
# compatible with any /RTC mode. Keep this override local to the instrumented
# library; parent targets and non-instrumented project tests retain their flags.
if(MSVC AND QT_ANT_DESIGN_ENABLE_ADDRESS_SANITIZER)
    foreach(_qt_ant_design_flag_var IN ITEMS CMAKE_C_FLAGS_DEBUG CMAKE_CXX_FLAGS_DEBUG)
        string(REGEX REPLACE "[/-]RTC[1csu]+" ""
            ${_qt_ant_design_flag_var} "${${_qt_ant_design_flag_var}}")
    endforeach()
    unset(_qt_ant_design_flag_var)
endif()

# Add Qt resource file
set(QT_ANT_DESIGN_RESOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/../resources/qt-ant-design.qrc
)

add_library(qt-ant-design
    ${QT_ANT_DESIGN_HEADERS}
    ${QT_ANT_DESIGN_SOURCES}
    ${QT_ANT_DESIGN_RESOURCES}
    ${QT_ANT_DESIGN_VERSION_HEADER}
)

get_target_property(QT_ANT_DESIGN_TARGET_TYPE qt-ant-design TYPE)
if(QT_ANT_DESIGN_TARGET_TYPE STREQUAL "STATIC_LIBRARY")
    target_compile_definitions(qt-ant-design PUBLIC QT_ANT_DESIGN_STATIC_DEFINE)
    if(MSVC)
        if(CMAKE_VERSION VERSION_GREATER_EQUAL "3.25")
            set_property(TARGET qt-ant-design PROPERTY
                MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>"
            )
        else()
            target_compile_options(qt-ant-design PRIVATE
                "$<$<CONFIG:Debug,RelWithDebInfo>:/Z7>"
            )
        endif()
    endif()
else()
    target_compile_definitions(qt-ant-design PRIVATE QT_ANT_DESIGN_LIBRARY)
endif()

target_include_directories(qt-ant-design
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
        $<BUILD_INTERFACE:${QT_ANT_DESIGN_GENERATED_INCLUDE_DIR}>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/qt-ant-design>
)

target_link_libraries(qt-ant-design
    PUBLIC
        Qt${QT_VERSION_MAJOR}::Core
        Qt${QT_VERSION_MAJOR}::Widgets
        Qt${QT_VERSION_MAJOR}::Svg
)

set(QT_ANT_DESIGN_SANITIZER_COMPILE_OPTIONS)
set(QT_ANT_DESIGN_SANITIZER_LINK_OPTIONS)
if(QT_ANT_DESIGN_ENABLE_ADDRESS_SANITIZER)
    if(MSVC)
        list(APPEND QT_ANT_DESIGN_SANITIZER_COMPILE_OPTIONS
            /fsanitize=address
        )
        list(APPEND QT_ANT_DESIGN_SANITIZER_LINK_OPTIONS /INCREMENTAL:NO)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        list(APPEND QT_ANT_DESIGN_SANITIZER_COMPILE_OPTIONS -fsanitize=address -fno-omit-frame-pointer)
        list(APPEND QT_ANT_DESIGN_SANITIZER_LINK_OPTIONS -fsanitize=address)
    else()
        message(FATAL_ERROR "AddressSanitizer is not configured for ${CMAKE_CXX_COMPILER_ID}")
    endif()
endif()
if(QT_ANT_DESIGN_ENABLE_UNDEFINED_SANITIZER)
    if(MSVC)
        message(FATAL_ERROR "UndefinedBehaviorSanitizer is not available with MSVC; use Clang or GCC")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        list(APPEND QT_ANT_DESIGN_SANITIZER_COMPILE_OPTIONS -fsanitize=undefined -fno-omit-frame-pointer)
        list(APPEND QT_ANT_DESIGN_SANITIZER_LINK_OPTIONS -fsanitize=undefined)
    else()
        message(FATAL_ERROR "UndefinedBehaviorSanitizer is not configured for ${CMAKE_CXX_COMPILER_ID}")
    endif()
endif()
if(QT_ANT_DESIGN_SANITIZER_COMPILE_OPTIONS)
    if(MSVC AND QT_ANT_DESIGN_ENABLE_ADDRESS_SANITIZER)
        # A build-tree parent may intentionally remain uninstrumented. Disable
        # MSVC STL container annotations inside the instrumented library so its
        # object mismatch tags stay compatible with that consumer; project code
        # remains fully AddressSanitizer-instrumented.
        target_compile_definitions(qt-ant-design PRIVATE
            _DISABLE_STRING_ANNOTATION
            _DISABLE_VECTOR_ANNOTATION
        )
    endif()
    foreach(sanitizer_option IN LISTS QT_ANT_DESIGN_SANITIZER_COMPILE_OPTIONS)
        # Instrument the library itself. Propagating compiler instrumentation to
        # arbitrary parent targets would combine MSVC /fsanitize=address with the
        # parent's default Debug /RTC1. Link requirements remain PUBLIC below.
        target_compile_options(qt-ant-design PRIVATE "$<BUILD_INTERFACE:${sanitizer_option}>")
    endforeach()
endif()
if(QT_ANT_DESIGN_SANITIZER_LINK_OPTIONS)
    foreach(sanitizer_option IN LISTS QT_ANT_DESIGN_SANITIZER_LINK_OPTIONS)
        target_link_options(qt-ant-design PUBLIC "$<BUILD_INTERFACE:${sanitizer_option}>")
    endforeach()
endif()
unset(sanitizer_option)
unset(QT_ANT_DESIGN_SANITIZER_COMPILE_OPTIONS)
unset(QT_ANT_DESIGN_SANITIZER_LINK_OPTIONS)

# Install headers preserving directory structure
install(DIRECTORY core styles widgets
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/qt-ant-design
    FILES_MATCHING PATTERN "*.h"
    PATTERN "*_p.h" EXCLUDE
    PATTERN "AntSelectPopup.h" EXCLUDE
)

install(FILES ${QT_ANT_DESIGN_VERSION_HEADER}
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/qt-ant-design/core
)
